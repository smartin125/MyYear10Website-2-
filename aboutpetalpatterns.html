from turtle import *
def draw_petal(colour):
pencolor(colour)
for i in range(4):
forward(60)
left(90)
pensize(5)
bgcolor("lightgreen")
fillcolor("hotpink")
petals = int(input("How many petals? "))
angle = 360/petals
begin_fill()
for i in range(petals):
draw_petal("purple")
left(angle)
end_fill()
